# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.27.0
engine.minor=29
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# The chamber is also played with the pointer or a finger: a tap on empty
# space spawns the selected material there, a tap on a ball selects it (in
# joint mode two taps connect two balls); every action has a tray button.
engine.group.cursor_up=Spawn cursor
control.cursor_up=ArrowUp+KeyW | Move cursor up
control.cursor_down=ArrowDown+KeyS | Move cursor down
control.cursor_left=ArrowLeft+KeyA | Move cursor left
control.cursor_right=ArrowRight+KeyD | Move cursor right
control.spawn=Enter+NumpadEnter+Space | Spawn material
engine.group.material_1=Build
control.material_1=Digit1+Numpad1 | Material 1
control.material_2=Digit2+Numpad2 | Material 2
control.material_3=Digit3+Numpad3 | Material 3
control.material_4=Digit4+Numpad4 | Material 4
control.joint=KeyJ | Joint mode
control.delete=Delete+Backspace | Remove selected body
control.undo=KeyU | Undo
engine.group.run=Chamber
control.run=KeyR | Run simulation
control.reframe=KeyC | Re-frame camera
control.pause=Escape | Pause / cancel

# Gamepad defaults (the browser build had none): the D-pad or left stick
# moves the spawn cursor, South spawns, the shoulders pick materials 1 and 2,
# North toggles joint mode, West deletes the selected body, the left trigger
# undoes, the right trigger runs, the right stick press re-frames the camera,
# Start or East pauses / cancels.
engine.gamepad.cursor_up=DpadUp+LeftStickUp
engine.gamepad.cursor_down=DpadDown+LeftStickDown
engine.gamepad.cursor_left=DpadLeft+LeftStickLeft
engine.gamepad.cursor_right=DpadRight+LeftStickRight
engine.gamepad.spawn=South
engine.gamepad.material_1=LeftShoulder
engine.gamepad.material_2=RightShoulder
engine.gamepad.joint=North
engine.gamepad.delete=West
engine.gamepad.undo=LeftTrigger
engine.gamepad.run=RightTrigger
engine.gamepad.reframe=RightStickPress
engine.gamepad.pause=Start+East
